\section{Hamiltonian and Fermion map}

\begin{frame}[allowframebreaks]{The $H_{\mathrm{hc}}$}
    \begin{theorem}[\cite{Kitaev2006}]
    Under a weak magnetic field of the form $\mathbf{h} = (h_x,h_y,h_z) $, we denote that for the magnetic-free honeycomb model, the vortex-free section has energy spectrum $E_i^{(0)}$ and the two-vortex section has energy spectrum $E_j^{(2)}$. Assumes that  $E_i^{(0)} - E_j^{(2)} \approx \Delta \geq |E^{(0)}_j - E^{(0)}_k|,|E^{(2)}_j - E^{(2)}_k|$ , then the \textit{effective Hamiltonian} writes:  
    \begin{align}
    H_{hc}=&-J_x'\sum_{(j,k)\mathrm{~is~}x-\mathrm{links}}\sigma_j^x\sigma_k^x-J_y'\sum_{(j,k)\mathrm{~is~}y-\mathrm{links}}\sigma_j^y\sigma_k^y\\
    &-J_z'\sum_{(j,k)\mathrm{~is~}z-\mathrm{links}}\sigma_j^z\sigma_k^z - K\sum_{j,k,l}\sigma_j^x \sigma_k^y \sigma_l^z,
    \end{align}
    where $K = {h_xh_yh_z}/{\Delta^2}$ and $J_{\alpha}' = J_{\alpha} + h_{\alpha}^2/\Delta$.
\end{theorem}
\begin{itemize}
    \item The magnetic field is used to generate the Energy gap.
    \item  Can also be generated by a system error within the Floquet simulation~\cite{evered2025probingtopologicalmatterfermion}.
    \item  $W_p := \sigma_1^x \sigma_2^y \sigma_3^z \sigma_4^x\sigma_5^y \sigma_6^z$ defined on an arbitrary hexagon $p:123456$ are conservatives
\end{itemize}

\begin{frame}[allowframebreaks]{The fermion map} 
\begin{theorem}
      Consider two fermion $\{f_j^{(1)}, f_j^{(2)}\}$  and their four-dimensional Fock space and operators $\mathcal{F}:=\bigotimes_{j = 1}^N\{\ket{},f_j^{(1),\dagger}\ket{}, f_j^{(2),\dagger}\ket{}, f_j^{(2),\dagger} f_j^{(1),\dagger}\ket{}\},$
    \begin{align}
         \begin{cases}b^x = f^{(1)}+f^{\dagger,(1)}, b^y = ( f^{(1)}-f^{\dagger,(1)})/i,\\b^z =f^{(2)}+f^{\dagger,(2)},c = ( f^{(2)}-f^{\dagger,(2)})/i,\\ D_j :=b_j^x b_j^y b_j^zc_j  .\end{cases}
    \end{align}
    Restrained on the subspace $\mathcal{M}:= \{\ket{v} ~|~\forall j, D_j\ket{v} =  \ket{v}\}$ and under the transformation $\sigma_j^{\alpha} \to \tilde{\sigma}_j^{\alpha}=i b_j^{\alpha}c_j$, the eigenvalue and eigenspaces of this transformed Majorana Fermion Hamiltonian are identical to those of the third-order magnetic perturbed honeycomb model.     
\end{theorem}
\newpage
  Use $\alpha_{jk} \in \{x,y,z\}$ to denote the edge type that links the adjacent points $j,k$. Define the notations $\hat u_{jk}:= i b_j^{\alpha_{jk}}b_k^{\alpha_{jk}}$ and 
   \begin{align}
       &J_{{jk}} = \begin{cases}J_x &\text{if (j,k) is x-links,}\\J_y &\text{if (j,k) is y-links,}\\J_z &\text{if (j,k) is z-links,}\\0 &\text{if else.}\end{cases}, \\
       &K_{jk} = \begin{cases}\text{sgn}(jk) \cdot h_x h_yh_z/\Delta^2 &\text{if (j,k) is Second-Nearest Neighbor,}\\0 &\text{if else.}\end{cases}
   \end{align}
   Here $\text{sgn}(jk) = +1$ is decided by whether $\alpha_{jl}\to \alpha_{kl} \to \beta_{l}$ has the same parity of $x \to y \to z$.
   We can prove the theorem below:
\begin{theorem}
   Restrained on the subspace $\mathcal{M}:= \{\ket{v} ~|~\forall j, D_j\ket{v} =  \ket{v}\}$  and under the transformation $\sigma_j^{\alpha} \to \tilde{\sigma}_j^{\alpha}=i b_j^{\alpha}c_j$, the third-order magnetic perturbed honeycomb Hamiltonian and its conserved operators $W_p$ are transformed into the Majorana Fermion Hamiltonian and operators according to
    \begin{align}
        \tilde H&= ({i}/{4})\cdot \sum_{j,k} (2J_{{jk}}\hat u_{jk} + 2K_{jk}\hat  u_{jl}\hat u_{kl}) c_j c_k,\label{eq:trans_Ham} \\
        \tilde W_p &= -\hat{u}_{12}\hat{u}_{23}\hat{u}_{34}\hat{u}_{45}\hat{u}_{56}\hat{u}_{61},\label{eq:conser_q}
    \end{align}  
    where $l$ is the unique mid-point of the second-nearest neighbor pair $(j,k)$.
\end{theorem}
\end{frame}

\begin{frame}{Transform to free Fermion}
Now we transform the \textbf{interactive Fermion model} above into \textbf{multiple free Fermion}.
\begin{align}\label{eq:free_ferm}
    \tilde H(\{u_{jk}\}) =  ({i}/{4})\cdot \sum_{j,k} A_{jk} c_j c_k, \quad A_{jk} = 2J_{{jk}}u_{jk} + 2K_{jk}u_{jl} u_{kl},
\end{align}
\begin{lemma}
    Consider a $\mathbb{Z}_2$ gauge field $\{u_{jk}\}$ and define its vortex field as $\{w_p\}_p$ where $w_p = u_{12}u_{23}u_{34}u_{45}u_{56}u_{61}$ is defined on an arbitrary hexagon $p:123456$. 
    Then for the eigenvectors $\ket{v}$ of $\tilde H(\{u_{jk}\})$ in Eq.~\eqref{eq:free_ferm} such that satisfy $\tilde H \ket{v} = \lambda_v \ket{v}$. Those $\ket{v}$ are one-to-one corresponding with those vectors such that $H_{\mathrm{hc}} \ket{\phi} = \lambda_{p} \ket{\phi}$and $W_p\ket{\phi} = -w_p \ket{\phi}$ . 
\end{lemma}
\end{frame}

\begin{frame}[allowframebreaks]{Solution of free Fermion}
We consider an arbitrary free Majorana system, where $\mathcal{H} =  \frac{i}{4}\sum_{j,k = 1}^{2N}A_{jk}c_jc_k$ and $A$ is an antisymmetric matrix. 
\begin{lemma}
\label{lemma:diag}
For any even-dimensional real skew-symmetric matrix $A$ (i.e., $A^T = -A$), there exists a real orthogonal matrix $Q$ (i.e., $Q^T Q = I$) such that $A$ can be decomposed into the following block-diagonal form where all $\epsilon_i$ are non-negative real numbers:
\begin{align}
    A = Q\Lambda(\varepsilon_i) Q^T = Q
\begin{pmatrix}
\begin{matrix} 0 & \epsilon_1 \\ -\epsilon_1 & 0 \end{matrix} & &  \\
& \ddots & \\
 & & \begin{matrix} 0 & \epsilon_k \\ -\epsilon_k & 0 \end{matrix} \\
\end{pmatrix}
Q^T.
\end{align}
\end{lemma}

\begin{theorem}[topological obstruction]\label{theo: topo_ob}
   Rewrites the $\mathcal{H}$ under the periodic condition as $\mathcal{H}_S$ with $A_{S},B_S$.   If it is impossible to find a cut-bipartite cut $\mathbf{n}$ such that it can generate a bipartite diagonalized matrix $B'=\mathcal{D}_{\mathbf{n}}\circ \mathcal{L}_{\mathbf n}(B_S)$ with $B'^2 = -\mathbb{I}$,  then the Majorana zero modes of $\mathcal{H}$ \textbf{exist}.
\end{theorem}

\begin{theorem}\label{theo:suff_cond_zeromode}
\begin{itemize}
    \item  If for an arbitrary cut $\mathbf{n}$, $\mathcal{M}_{\mathbf{n}}(B_S) = -1$, then the Majorana zero modes\textbf{ exist} within $\mathcal{H}$.
    \item  $ \mathcal{M}(B,B') :=\mathcal{M}_{\mathbf{n}}(B) \mathcal{M}_{\mathbf{n}}(B')= (-1)^{\nu(P)}$, where $\nu(P) = 2\pi i \text{Tr}\left([P \Pi^{\mathbf e_x} P,P \Pi^{\mathbf e_y} P]\right)$.
\end{itemize}
\end{theorem}
\begin{lemma}\label{lemma:Chern_numb_cal}
    Given the Fourier transformation of a translationally invariant projective matrix $P$, its Chern number $\nu(P)$ can be calculated by
    \begin{align}
        \nu(P) = \frac{1}{2\pi i} \int dq_xdq_y \text{Tr}\left(\tilde P(\mathbf{q})\left(\frac{\partial\tilde P(\mathbf{q})}{\partial q_x}\frac{\partial\tilde P(\mathbf{q})}{\partial q_y}-\frac{\partial\tilde P(\mathbf{q})}{\partial q_x}\frac{\partial\tilde P(\mathbf{q})}{\partial q_y}\right)\right).
    \end{align}
\end{lemma}

\chapter{The spectrum analysis of the 2d-TFIM}



\begin{theorem}
Given a Hamiltonian with the form $H(\Omega) = -\Omega T + D$ with spectrum structure denoted as $E_n(\Omega)$, where $D$ is diagonal in the computational basis, and $T = \sum_{j} X_j$ is globally uniform transversal field. If the spectrum of $D$ satisfied [tbc], then there exists $\Omega^*$ such that $E_1(\Omega) - E_0(\Omega) \leq O(e^{-n})$, where $n$ is the [tbc].
\end{theorem}
